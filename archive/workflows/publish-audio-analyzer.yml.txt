name: build pm-audio-analyzer

# Triggers only on changes inside the audio-analyzer folder or this
# workflow itself. Image is ~3-4 GB (PyTorch-CUDA base + transformers +
# autoawq prebuilt wheel). No flash-attn-from-source compile, so the
# build is much faster than pm-qwen-tts: ~8-12 min first build, ~30-60s
# on a cache hit.
on:
  push:
    branches: [main]
    tags: ["audio-analyzer-v*"]
    paths:
      - "pm-audio-analyzer/**"
      - ".github/workflows/publish-audio-analyzer.yml"
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    # No source compile here, but the CUDA-devel base + torch are big.
    # Cap at 30 min — a normal first build is ~10 min.
    timeout-minutes: 30
    permissions:
      contents: read
      packages: write

    steps:
      # The default GHA runner has only ~14 GB free — pytorch-devel +
      # torch + autoawq layers fit only after we clear pre-installed
      # dotnet/swift/android/haskell baggage. Same trick as the qwen-tts
      # workflow.
      - name: Free disk space
        uses: jlumbroso/free-disk-space@main
        with:
          tool-cache: true
          android: true
          dotnet: true
          haskell: true
          large-packages: true
          docker-images: false  # we need these in a moment
          swap-storage: true

      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up Buildx
        uses: docker/setup-buildx-action@v3

      - name: Login to ghcr.io
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Image metadata
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ghcr.io/theingof/pm-audio-analyzer-dev
          tags: |
            type=ref,event=branch
            type=ref,event=tag
            type=sha,prefix=,format=short
            type=raw,value=latest,enable=${{ github.ref == 'refs/heads/main' }}

      - name: Build and push
        uses: docker/build-push-action@v5
        with:
          context: ./pm-audio-analyzer
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          cache-from: type=gha
          cache-to: type=gha,mode=max
